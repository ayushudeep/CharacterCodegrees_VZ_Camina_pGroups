My2A := function(p)

F<a, a1, a2> := FreeGroup (3);
Alpha := [a, a1];
Beta := [a2];

// common relations, typically commutators
Comms := [
(a1, a) = a2, a2^p = 1] cat
&cat([[(Beta[i], Alpha[j]) = Id(F) : j
 in [1..2]]: i in [1]]);

L := [];

// (31)

G := quo <F | Comms, a^(p^2) = a2, a1^p = 1>;
Append (~L, G);

// (22)

G := quo <F | Comms, a^(p) = a2, a1^(p^2) = 1>;
Append (~L, G);

// (211)c

G := quo <F | Comms, a^(p^2) = 1, a1^p = 1>;
Append (~L, G);

return L;
end function;

My2B := function(p)

F<a, a1, a2, ga> := FreeGroup (4);
Alpha := [a, a1, a2];
Beta := [ga];

// common relations, typically commutators
Comms := [
(a1, a) = a2, (a2, a)= 1, (a1, a2) = 1, a2^p = 1] cat
&cat([[(Beta[i], Alpha[j]) = Id(F) : j
 in [1..3]]: i in [1]]);

L := [];

// (211)a

G := quo <F | Comms, a^(p) = a2, a1^(p) = 1, ga^(p) = 1>;
Append (~L, G);

// (1^4)

G := quo <F | Comms, a^(p) = 1, a1^(p) = 1, ga^(p) = 1>;
Append (~L, G);

// (211)b

G := quo <F | Comms, a^(p) = 1, a1^(p) = 1, ga^(p) = a2>;
Append (~L, G);


return L;
end function;

My2 := function(p)
return My2A(p) cat My2B(p);
end function;

  end for;
