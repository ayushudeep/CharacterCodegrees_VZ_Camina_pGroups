
My2A := function(p)

K := GF(p);

w := PrimitiveRoot(p);

NonQuadraticResidue := function (p)
  for x in GF (p) do
    if not IsPower (x, 2) then return x; end if;
  end for;
end function;

Z := Integers ();
    nu := Z ! (NonQuadraticResidue (p)); // nu

F<a, a1, a2> := FreeGroup (3);
Alpha := [a, a1];
Beta := [a2];

// common relations, typically commutators
Comms := [
(a1, a) = a2, a2^p = 1] cat
&cat([[(Beta[i], Alpha[j]) = Id(F) : j
 in [1..2]]: i in [1]]);

L := [];

// (41)

G := quo <F | Comms, a^(p^3) = a2, a1^p = 1>;
Append (~L, G);

// (32)a_1

G := quo <F | Comms, a^(p^2) = a2, a1^(p^2) = 1>;
Append (~L, G);

// (32)a_2

G := quo <F | Comms, a^(p^3) = 1, a1^p = a2>;
Append (~L, G);

// (311)c

G := quo <F | Comms, a^(p^3) = 1, a1^p = 1>;
Append (~L, G);

// (221)d

G := quo <F | Comms, a^(p^2) = 1, a1^(p^2) = 1>;
Append (~L, G);

return L;
end function;
